cmake_minimum_required(VERSION 3.25)
project(SandVnc LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(third_party/minhook)
add_library(sandvnc-input SHARED native/adapter.cpp)
target_link_libraries(sandvnc-input PRIVATE minhook user32)
target_compile_definitions(sandvnc-input PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(input-probe WIN32 native/probe.cpp)
target_link_libraries(input-probe PRIVATE user32 gdi32)
target_compile_definitions(input-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(client-launcher native/client-launcher.cpp)
target_compile_definitions(client-launcher PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
